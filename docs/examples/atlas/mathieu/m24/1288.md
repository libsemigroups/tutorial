## Claim

Let $u$ and $v$ be the two subgroup words listed in the code below. If

$$
    H = \langle u,v \rangle \cong \mathrm{M}_{12}:2,
$$

then

$$
    [\mathrm{M}_{24}:H] = 1,288.
$$

We verify the claimed index, but not the claimed isomorphism type. In the code,
uppercase letters denote inverses: `A` represents $a^{-1}$ and `B` represents
$b^{-1}$.

## The code

The following script includes the full presentation and can be run on its own.

<details markdown>
<summary>Code</summary>

```python
from datetime import timedelta

from libsemigroups_pybind11 import Presentation, ToddCoxeter, congruence_kind, presentation
from libsemigroups_pybind11.words import parse_relations

# Enable the empty word and inverses so the presentation defines a group.
p = Presentation("abAB")
p.contains_empty_word(True)
presentation.add_inverse_rules(p, "ABab")

# Add the defining relations
presentation.add_rule(p, parse_relations("a^2"), "")
presentation.add_rule(p, parse_relations("b^3"), "")
presentation.add_rule(p, parse_relations("(ab)^23"), "")
presentation.add_rule(p, parse_relations("(ABab)^12"), "")
# [a, bab] = a^-1 (bab)^-1 a (bab).
presentation.add_rule(p, parse_relations("(ABABabab)^5"), "")
presentation.add_rule(p, parse_relations("(ababaB)^3(abaBaB)^3"), "")
# [a, b^-1 a b a b]^5 = 1, marked redundant in the ATLAS source.
presentation.add_rule(p, parse_relations("(ABABAbaBabab)^5"), "")
presentation.add_rule(p, parse_relations("(ab(abaB)^3)^4"), "")

presentation.replace_word_with_new_generator(p, presentation.longest_subword_reducing_length(p))
presentation.replace_word_with_new_generator(p, presentation.longest_subword_reducing_length(p))
presentation.sort_each_rule(p)
presentation.sort_rules(p)

# M12.2, order 190080, expected index 1288.
# Source: https://brauer.maths.qmul.ac.uk/Atlas/spor/M24/words/M24G1-max4W1
# ATLAS output words simplified using a^2 = b^3 = 1; B denotes b^-1.
tc = ToddCoxeter(congruence_kind.onesided, p)
# Felsch settings from the original m24.py enumeration.
tc.strategy(tc.options.strategy.felsch)
tc.def_version(tc.options.def_version.two)
tc.def_policy(tc.options.def_policy.discard_all_if_no_space)
tc.use_relations_in_extra(True)
tc.def_max(10000)
tc.lower_bound(1288)
for word in (
    "a",
    "baBabaBabaBabaBabaBaBaBaBaBaBaBaBaBaBaBaBaBababababababababababababababaBaba"
    "BabaBabaBabaB",
):
    tc.add_generating_pair(word, "")

tc.report_every(timedelta(seconds=1))

actual_index = tc.number_of_classes()
print(f"The index of M12.2 is {actual_index}")
if actual_index != 1288:
    raise RuntimeError(f"expected index 1288, got {actual_index}")
```
